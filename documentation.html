DMI ID

DMI ID — это система авторизации, основанная на подтверждении входа через мобильное устройство с помощью свайпа. Она использует MQTT-брокер для обмена данными в реальном времени и не требует сложных серверных настроек.

Ключевая особенность: Ваш код автоматически генерирует уникальный ID сессии и ссылку для входа. Пользователь открывает её на телефоне, подтверждает вход, а ваше приложение мгновенно получает его имя и номер телефона.

Как это работает

  1. Ваше приложение генерирует случайный sessionId (например, A1B2C3D4).
  2. Приложение формирует ссылку вида auth.html?session=A1B2C3D4&app=Название.
  3. Приложение подписывается на MQTT-топик dmi/auth/session/A1B2C3D4.
  4. Пользователь открывает ссылку, вводит данные и свайпает.
  5. Сайт отправляет в топик JSON: {"status": "success", "phone": "+7999...", "name": "Иван"}.
  6. Ваше приложение ловит это сообщение и авторизует пользователя.

Безопасность

Система DMI ID защищает пользователей от фишинга. В ссылке вы передаете название вашего приложения в параметре app.

  • Если ваше приложение добавлено в официальный реестр DMI, пользователь увидит название приложения без предупреждений.
  • Если приложение неизвестно системе (сторонняя разработка), сайт auth.html автоматически покажет красное предупреждение: «Не рекомендуем входить в это приложение». Это нормальное поведение для сторонних сервисов.

JavaScript (Frontend)

Пример реализации кнопки входа для веб-сайта с использованием MQTT через WebSockets. Код автоматически генерирует ссылку.

// 1. Подключаем MQTT клиент
const mqttClient = mqtt.connect('wss://broker.emqx.io:8084/mqtt');

// 2. Генерируем случайный ID сессии
const sessionId = Math.random().toString(36).substring(2, 10).toUpperCase();

// 3. Автоматически формируем ссылку
const appName = "My Super App";
const link = `http://pasport-dmi.cc.cd/auth.html?session=${sessionId}&app=${encodeURIComponent(appName)}`;

console.log("Отправьте пользователю эту ссылку:", link);

// Выводим ссылку на экран
document.getElementById('login-btn').onclick = () => {
    window.open(link, '_blank');
};

// 4. Подписываемся на топик этой сессии и ждем ответа
mqttClient.on('connect', () => {
    mqttClient.subscribe(`dmi/auth/session/${sessionId}`);
});

mqttClient.on('message', (topic, message) => {
    const data = JSON.parse(message.toString());
    if (data.status === 'success') {
        console.log("Пользователь вошел!");
        console.log("Имя:", data.name);
        console.log("Телефон:", data.phone);
        
        // Авторизуем пользователя в вашей системе
        localStorage.setItem('token', data.phone);
    }
});

Python (Backend / Telegram Bot)

Пример на Python, который сам генерирует уникальный код и ссылку, а затем ожидает подтверждения.

import paho.mqtt.client as mqtt
import json
import random
import string

# 1. Генерируем случайный ID сессии (8 символов)
session_id = ''.join(random.choices(string.ascii_uppercase + string.digits, k=8))
app_name = "My Super App"

# 2. Автоматически формируем ссылку
link = f"http://pasport-dmi.cc.cd/auth.html?session={session_id}&app={app_name}"
print(f"Отправьте пользователю ссылку: {link}")

topic = f"dmi/auth/session/{session_id}"

def on_connect(client, userdata, flags, rc):
    print("Ожидание входа...")
    client.subscribe(topic)

def on_message(client, userdata, msg):
    data = json.loads(msg.payload.decode())
    if data.get("status") == "success":
        print(f"✅ Успешный вход!")
        print(f"👤 Имя: {data.get('name')}")
        print(f"📱 Телефон: {data.get('phone')}")
        client.disconnect()

client = mqtt.Client()
client.on_connect = on_connect
client.on_message = on_message

client.connect("broker.emqx.io", 1883, 60)
client.loop_forever()

Node.js

Для серверной части на Node.js. Использует встроенный модуль crypto для генерации сессии.

const mqtt = require('mqtt');
const crypto = require('crypto');

const client = mqtt.connect('mqtt://broker.emqx.io:1883');

// 1. Генерируем случайный ID
const sessionId = crypto.randomBytes(4).toString('hex').toUpperCase();
const appName = encodeURIComponent("My Super App");

// 2. Формируем ссылку
const link = `http://pasport-dmi.cc.cd/auth.html?session=${sessionId}&app=${appName}`;
console.log("Ссылка для входа:", link);

const topic = `dmi/auth/session/${sessionId}`;

// 3. Ждем подтверждения
client.on('connect', () => {
    client.subscribe(topic, (err) => {
        if (!err) console.log(`Ожидание подтверждения...`);
    });
});

client.on('message', (topic, message) => {
    const data = JSON.parse(message.toString());
    
    if (data.status === 'success') {
        console.log('Успешный вход:', data.name);
        // Создаем JWT токен или сессию
    }
});

PHP

Пример на PHP с автоматической генерацией уникального ID и ссылки.

<?php
require('phpMQTT.php');

$server = 'broker.emqx.io';
$port = 1883;
$client_id = 'phpClient' . rand();

// 1. Генерируем случайный ID сессии
$sessionId = strtoupper(bin2hex(random_bytes(4)));
$appName = urlencode("My Super App");

// 2. Формируем ссылку
$link = "http://pasport-dmi.cc.cd/auth.html?session=" . $sessionId . "&app=" . $appName;
echo "Ссылка для входа: " . $link . "\n";

$mqtt = new phpMQTT($server, $port, $client_id);
$topic = "dmi/auth/session/" . $sessionId;

if ($mqtt->connect(true, NULL, "", "")) {
    $mqtt->subscribe($topic, 0);
    
    // Слушаем 30 секунд
    $startTime = time();
    while ($mqtt->proc() && (time() - $startTime < 30)) {
        // Ждем...
    }
    
    $mqtt->close();
}

function procmsg($topic, $msg){
    $data = json_decode($msg, true);
    if (isset($data['status']) && $data['status'] == 'success') {
        echo "Успешный вход!\n";
        echo "Имя: " . $data['name'] . "\n";
        echo "Телефон: " . $data['phone'] . "\n";
    }
}
?>

C++ (Arduino / ESP32)

Пример для умных устройств. Код генерирует случайный ID при запуске и формирует ссылку для вывода в Serial-порт (или на экран).

#include <WiFi.h>
#include <PubSubClient.h>

const char* ssid = "YOUR_WIFI";
const char* password = "YOUR_PASS";
const char* mqtt_server = "broker.emqx.io";

WiFiClient espClient;
PubSubClient client(espClient);

String sessionId = "";
String topic = "";

// Функция генерации случайного ID
String generateSessionId() {
  String str = "";
  const char chars[] = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
  for (int i = 0; i < 8; i++) {
    str += chars[random(0, 36)];
  }
  return str;
}

void callback(char* topic, byte* payload, unsigned int length) {
  String message = "";
  for (int i = 0; i < length; i++) {
    message += (char)payload[i];
  }
  
  if (message.indexOf("\"status\":\"success\"") > 0) {
    Serial.println("Успешный вход!");
    // Тут можно включить замок или зажечь свет
  }
}

void setup() {
  Serial.begin(115200);
  
  // 1. Генерируем ID и ссылку
  sessionId = generateSessionId();
  topic = "dmi/auth/session/" + sessionId;
  String appName = "My%20Super%20App"; // URL-encoded
  String link = "http://pasport-dmi.cc.cd/auth.html?session=" + sessionId + "&app=" + appName;
  
  Serial.println("Отправьте пользователю ссылку:");
  Serial.println(link);
  
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  
  client.setServer(mqtt_server, 1883);
  client.setCallback(callback);
  
  // 2. Подписываемся на созданный топик
  if (client.connect("ESP32Client")) {
    client.subscribe(topic.c_str());
  }
}

void loop() {
  if (!client.connected()) {
    // Реконнект...
  }
  client.loop();
}
